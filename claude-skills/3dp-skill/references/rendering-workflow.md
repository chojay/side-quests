## Primary: Interactive HTML Viewer

The preferred visualization method is the interactive 3D HTML viewer generated by `scripts/export_with_viewer.py`. It produces a self-contained HTML file with embedded STL data rendered via three.js. No additional dependencies required.

```python
# As module
from export_with_viewer import export_with_viewer
export_with_viewer(part.part, "my_part", dimensions="50x40x30mm")

# As CLI
# python scripts/export_with_viewer.py model.stl --name "My Part"
```

The viewer template lives at `assets/viewer_template.html` and supports rotation, zoom, and pan.

---

## Optional: Static Image Previews (matplotlib)

For static PNG/JPEG previews (useful for documentation or README files), use the matplotlib-based renderer. This requires additional dependencies: `pip install numpy-stl matplotlib numpy`.

### Quick Start

```python
from render_stl_preview import render_stl_previews

# Generate top-down and isometric views
results = render_stl_previews("model.stl")
# Creates: model_top.png, model_isometric.png
```

## Command Line Usage

```bash
# Basic usage (top + isometric views)
python render_stl_preview.py model.stl

# All views (top, isometric, front, side, combined)
python render_stl_preview.py model.stl --views all

# Custom output directory
python render_stl_preview.py model.stl --output-dir ./previews

# JPEG format at higher resolution
python render_stl_preview.py model.stl --format jpeg --dpi 300
```

## Available Views

| View | Angle | Best For |
|------|-------|----------|
| **Top** | Looking down Z axis | Floor plan, XY layout |
| **Isometric** | 35° elev, 45° azim | Overall shape, 3D understanding |
| **Front** | Looking at XZ plane | Height profile, front features |
| **Side** | Looking at YZ plane | Depth profile, side features |
| **Combined** | 2x2 grid of all views | Complete overview |

## Integration in Templates

Add to any template's `main()` function:

```python
def main():
    # ... generate part ...
    part.part.export_stl("output.stl")

    # Generate preview images
    try:
        from render_stl_preview import render_stl_previews
        previews = render_stl_previews("output.stl", views='basic')
        print(f"✓ Preview images generated")
    except ImportError:
        print("Note: Install numpy-stl matplotlib for preview images")
```

## Inline Rendering Function

For standalone use without the reference script:

```python
import numpy as np
from stl import mesh
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d.art3d import Poly3DCollection


def render_stl_to_image(stl_path, output_path, view='isometric', dpi=150):
    """
    Render STL to image file.

    Args:
        stl_path: Path to STL file
        output_path: Path for output image (png/jpeg)
        view: 'top', 'isometric', 'front', or 'side'
        dpi: Image resolution
    """
    # Load mesh
    stl_mesh = mesh.Mesh.from_file(stl_path)

    # Get bounds
    vectors = stl_mesh.vectors
    all_points = vectors.reshape(-1, 3)
    min_pt = all_points.min(axis=0)
    max_pt = all_points.max(axis=0)
    center = (min_pt + max_pt) / 2
    max_range = (max_pt - min_pt).max() / 2

    # Create figure
    fig = plt.figure(figsize=(8, 8), facecolor='white')
    ax = fig.add_subplot(111, projection='3d')

    # Add mesh
    collection = Poly3DCollection(vectors, alpha=0.9)
    collection.set_facecolor('#4A90D9')
    collection.set_edgecolor('#2C5F8A')
    collection.set_linewidth(0.1)
    ax.add_collection3d(collection)

    # Set view angle
    view_angles = {
        'top': (90, -90),
        'isometric': (35, 45),
        'front': (0, -90),
        'side': (0, 0),
    }
    elev, azim = view_angles.get(view, (35, 45))
    ax.view_init(elev=elev, azim=azim)

    # Set limits
    ax.set_xlim(center[0] - max_range, center[0] + max_range)
    ax.set_ylim(center[1] - max_range, center[1] + max_range)
    ax.set_zlim(center[2] - max_range, center[2] + max_range)
    ax.set_box_aspect([1, 1, 1])

    # Labels
    ax.set_xlabel('X (mm)')
    ax.set_ylabel('Y (mm)')
    ax.set_zlabel('Z (mm)')
    ax.set_title(f'{view.title()} View')

    # Save
    plt.tight_layout()
    plt.savefig(output_path, dpi=dpi, bbox_inches='tight', facecolor='white')
    plt.close()

    return output_path


# Usage
render_stl_to_image("model.stl", "model_iso.png", view='isometric')
render_stl_to_image("model.stl", "model_top.png", view='top')
```

## Build123d Direct Rendering

For direct rendering from build123d without intermediate STL:

```python
from build123d import *
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d.art3d import Poly3DCollection
import numpy as np


def render_build123d_part(part, output_path, view='isometric', dpi=150):
    """
    Render build123d part directly to image.

    Args:
        part: build123d Part object
        output_path: Output image path
        view: 'top', 'isometric', 'front', 'side'
        dpi: Resolution
    """
    # Export to temporary STL, then render
    import tempfile
    import os

    with tempfile.NamedTemporaryFile(suffix='.stl', delete=False) as tmp:
        tmp_path = tmp.name

    try:
        part.export_stl(tmp_path)
        render_stl_to_image(tmp_path, output_path, view=view, dpi=dpi)
    finally:
        os.unlink(tmp_path)


# Usage with build123d
with BuildPart() as part:
    Box(50, 40, 30)
    # ... operations ...

render_build123d_part(part.part, "preview_iso.png", view='isometric')
render_build123d_part(part.part, "preview_top.png", view='top')
```

## Troubleshooting

### "Display not found" Error
Matplotlib may need a non-GUI backend for headless rendering:
```python
import matplotlib
matplotlib.use('Agg')  # Set before importing pyplot
import matplotlib.pyplot as plt
```

### Large STL Files
For models with >100k triangles, rendering may be slow:
```python
# Simplify mesh first
import trimesh
mesh = trimesh.load('large_model.stl')
simplified = mesh.simplify_quadric_decimation(50000)  # 50k faces
simplified.export('simplified.stl')
```

### Memory Issues
Close figures after saving:
```python
plt.savefig(output_path, ...)
plt.close()  # Important!
```

## Integration with Validation

Complete workflow in templates:

```python
def export_with_preview(part, base_name, validate=True):
    """Export STL with validation and preview images."""
    stl_path = f"{base_name}.stl"

    # Export STL
    part.export_stl(stl_path)
    print(f"✓ Exported: {stl_path}")

    # Validate
    if validate:
        import trimesh
        mesh = trimesh.load(stl_path)
        if mesh.is_watertight:
            print(f"✓ Validation: watertight")
        else:
            print(f"✗ Warning: not watertight")

    # Generate previews
    try:
        from render_stl_preview import render_stl_previews
        render_stl_previews(stl_path, views='basic')
    except ImportError:
        print("  (install numpy-stl matplotlib for previews)")

    return stl_path
```
